118 /*
119 @brief return the number of mS that the debounced input has been in true state
120
121 Note that this value starts incrementing after the debounce period, not after the physical pin transition.
122
123 @return returns a value in mS which saturates at 65535. Returns 0 if currently false.
124 */
125 uint16_t readDurationInTrueState_mS()
126 {
127
128 uint8_t tx[8] = { 201,_pin,_pinMode,1,0x55,0x55,0x55,0x55 };
129 uint8_t rx[8];
130 _sw.sendPacket(tx,rx);
131
132 transitions += (256 * rx[5] + rx[4]);
133 if (rx[3] == 0)
134 {
135 return (0);
136 }
137 else
138 {
139 return(256 * rx[7] + rx[6]);
140 }
141 }

316 void begin(int32_t* variableToIncrement,
317 int32_t slowIncrement = 1, uint32_t slow_mS_betweenIncrements = 250,
318 uint16_t slowToMediumTransition_mS = 1000,
319 int32_t mediumIncrement = 1, uint32_t medium_mS_betweenIncrements = 100,
320 uint16_t mediumToFastTransition_mS = 1000 ,
321 int32_t fastIncrement = 1, uint32_t fast_mS_betweenIncrements = 50)
322 {
323 _variableToIncrement = variableToIncrement;
324
325 _slowIncrement = slowIncrement;
326 _slow_mS_betweenIncrements = slow_mS_betweenIncrements;

328 _slowToMediumTransition_mS = slowToMediumTransition_mS;

337

345 bool update()
346 {
347 uint16_t pressDuration = _debouncedInput.readDurationInTrueState_mS();
348 int increments = 0;
349 bool incremented = false;
350 bool pressed = false;
351 if (pressDuration > 0)
352 {
353 if (_lastPressDuration >= pressDuration)
354 {
355 _lastPressDuration = 0;
356 }
357
358 if (pressDuration > _mediumToFastTransistion_mS)
359 {
360 // Increment fast
361 increments = (pressDuration - _lastPressDuration) / _fast_mS_betweenIncrements;
362 *_variableToIncrement += _fastIncrement * increments;
363 _lastPressDuration += _fast_mS_betweenIncrements * increments;
364 }
365 else if (pressDuration > _slowToMediumTransition_mS)
366 {
367 // Increment medium
368 increments = (pressDuration - _lastPressDuration) / _medium_mS_betweenIncrements;
369 *_variableToIncrement += _mediumIncrement * increments;
370 _lastPressDuration += _medium_mS_betweenIncrements * increments;
371 }
372 else
373 {
374 //Increment slow
375 increments = (pressDuration - _lastPressDuration) / _slow_mS_betweenIncrements;
376 *_variableToIncrement += _slowIncrement * increments;
377 _lastPressDuration += _slow_mS_betweenIncrements * increments;
378 incremented = increments > 0; // An increment happened
379 }
380 if (incremented)
381 {
382 _debouncedInput.transitions = 0; // Get rid of false->true transition so that final release doesn't cause and increment
383 }
384 pressed = true;
385 }
386 else
387 {
388 // Button isn't currently pressed. if there were other transitions, add them
389 _lastPressDuration = 0;
390 int presses = _debouncedInput.transitions / 2;
391 *_variableToIncrement += _slowIncrement * presses;
392 _debouncedInput.transitions -= presses * 2;
393 }
394
395 if (*_variableToIncrement > highLimit)
396 {
397 *_variableToIncrement = highLimit;
398 }
399 if (*_variableToIncrement < lowLimit)
400 {
401 *_variableToIncrement = lowLimit;
402 }
403
404 return (pressed);
405 }
406
408 long highLimit = LONG_MAX;
410 long lowLimit = LONG_MIN;
411
412private:
413 SerialWombatAbstractButton& _debouncedInput;

415

417 uint32_t _slow_mS_betweenIncrements;
